/* ==================== FILE: array_assign.c ==================== */

#include <stdio.h>

int main() {
    int i;
    int intArr[3];
    float floatArr[3];
    char charArr[3];

    // Integer array input
    printf("Enter 3 integer elements:\n");
    for(i = 0; i < 3; i++) {
        scanf("%d", &intArr[i]);
    }

    // Float array input
    printf("Enter 3 float elements:\n");
    for(i = 0; i < 3; i++) {
        scanf("%f", &floatArr[i]);
    }

    // Character array input (notice the space before %c to catch newline characters)
    printf("Enter 3 character elements:\n");
    for(i = 0; i < 3; i++) {
        scanf(" %c", &charArr[i]);
    }

    printf("\n--- Values and Memory Addresses ---\n");

    printf("\nInteger Array:\n");
    for(i = 0; i < 3; i++) {
        printf("Element %d -> Value: %d, Address: %p\n", i, intArr[i], (void*)&intArr[i]);
    }

    printf("\nFloat Array:\n");
    for(i = 0; i < 3; i++) {
        printf("Element %d -> Value: %.2f, Address: %p\n", i, floatArr[i], (void*)&floatArr[i]);
    }

    printf("\nCharacter Array:\n");
    for(i = 0; i < 3; i++) {
        printf("Element %d -> Value: %c, Address: %p\n", i, charArr[i], (void*)&charArr[i]);
    }

    return 0;
}

/* ==================== FILE: array_sum.c ==================== */

#include <stdio.h>

int main() {
    int n, i, sum = 0;

    printf("Enter the number of elements in the array: ");
    scanf("%d", &n);

    int arr[n]; // Variable-length array for dynamic sizing

    printf("Enter %d elements:\n", n);
    for(i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i]; // Add each element to the sum as we scan it
    }

    printf("The sum of all elements in the array is: %d\n", sum);

    return 0;
}

/* ================== END FILE: array_sum.c ================== */

/* ==================== FILE: calc.c ==================== */

#include <stdio.h>
int main() 
{
    int choice;
    double num1, num2;    
    do {
        printf("\n--- Calculator Menu ---\n");
        printf("1. Addition\n");
        printf("2. Subtraction\n");
        printf("3. Multiplication\n");
        printf("4. Division\n");
        printf("5. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice >= 1 && choice <= 4) 
        {
            printf("Enter two numbers: ");
            scanf("%lf %lf", &num1, &num2);
        }

        switch (choice) {
            case 1:
                printf("Result: %.2lf + %.2lf = %.2lf\n", num1, num2, num1 + num2);
                break;
            case 2:
                printf("Result: %.2lf - %.2lf = %.2lf\n", num1, num2, num1 - num2);
                break;
            case 3:
                printf("Result: %.2lf * %.2lf = %.2lf\n", num1, num2, num1 * num2);
                break;
            case 4:
                if (num2 != 0)
                    printf("Result: %.2lf / %.2lf = %.2lf\n", num1, num2, num1 / num2);
                else
                    printf("Error: Division by zero is not allowed.\n");
                break;
            case 5:
                printf("Exiting calculator. Goodbye!\n");
                break;
            default:
                printf("Invalid choice! Please choose between 1 and 5.\n");
        }
    } while (choice != 5);

    return 0;
}

/* ==================== FILE: factorial.c ==================== */

#include <stdio.h>
int main() {
    int n, i;
    unsigned long long fact = 1;
    printf("Enter an integer: ");
    scanf("%d", &n);

    // shows error if the user enters a negative integer
    if (n < 0)
        printf("Error! Factorial of a negative number doesn't exist.");
    else {
        for (i = 1; i <= n; ++i) {
            fact *= i;
        }
        printf("Factorial of %d = %llu", n, fact);
    }

    return 0;
}

/* ================== END FILE: factorial.c ================== */

/* ==================== FILE: swapping.c ==================== */

#include <stdio.h>
int main() 
{
    int a, b, temp;
    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);   
    printf("Before swapping: a = %d, b = %d\n", a, b);
    temp = a;
    a = b;
    b = temp;
    printf("After swapping: a = %d, b = %d\n", a, b);

    return 0;
}

/* ================== END FILE: swapping.c ================== */
